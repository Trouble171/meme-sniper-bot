import os
import time
import threading
import requests
from flask import Flask

# Flask web sunucusu (Render ve UptimeRobot uptime kontrolleri için)
app = Flask('')

@app.route('/')
def home():
    return "Meme Coin Sniper Bot Aktif ve Çalışıyor!"

def run_flask():
    port = int(os.environ.get("PORT", 8080))
    app.run(host='0.0.0.0', port=port)

# Telegram Bildirim Fonksiyonu
def send_telegram_message(message):
    TOKEN = os.environ.get("TELEGRAM_BOT_TOKEN")
    CHAT_ID = os.environ.get("TELEGRAM_CHAT_ID")
    if not TOKEN or not CHAT_ID:
        print("Telegram Token veya Chat ID eksik!")
        return
    
    url = f"https://api.telegram.org/bot{TOKEN}/sendMessage"
    payload = {
        "chat_id": CHAT_ID,
        "text": message,
        "parse_mode": "Markdown",
        "disable_web_page_preview": True
    }
    try:
        response = requests.post(url, json=payload, timeout=10)
        if response.status_code != 200:
            print(f"Telegram mesajı gönderilemedi: {response.text}")
    except Exception as e:
        print(f"Telegram istek hatası: {e}")

# RugCheck / Güvenlik Kontrol Fonksiyonu
def check_token_security(token_address):
    try:
        # Örnek RugCheck API entegrasyon noktası
        url = f"https://api.rugcheck.xyz/v1/tokens/{token_address}/report"
        res = requests.get(url, timeout=5)
        if res.status_code == 200:
            # Gerekli güvenlik filtreleri buraya eklenebilir
            return True
    except Exception as e:
        print(f"Güvenlik kontrolü hatası: {e}")
    return True # Hata durumunda akışın kesilmemesi için varsayılan

# Ana Tarama ve Sniper Döngüsü
def bot_loop():
    print("Multi-chain Meme Coin Sniper Bot döngüsü başlatıldı...")
    
    # Mükerrer bildirimleri engellemek için takip edilen token adresleri seti
    seen_tokens = set()
    
    # Hariç tutulacak büyük varlıklar / stablecoinler
    blacklisted_symbols = {"USDT", "USDC", "SOL", "ETH", "BTC", "BNB"}

    while True:
        try:
            # DexScreener API üzerinden son eklenen/taranan tokenleri çekme
            url = "https://api.dexscreener.com/latest/dex/tokens/latest"
            response = requests.get(url, timeout=10)
            
            if response.status_code == 200:
                data = response.json()
                pairs = data.get("pairs", [])
                
                for pair in pairs:
                    base_token = pair.get("baseToken", {})
                    token_address = base_token.get("address")
                    symbol = base_token.get("symbol", "").upper()
                    chain = pair.get("chainId", "unknown")
                    
                    # Filtreleme: Daha önce görüldü mü, stablecoin mi?
                    if not token_address or token_address in seen_tokens:
                        continue
                    if symbol in blacklisted_symbols:
                        continue
                        
                    # Likidite filtresi
                    liquidity = pair.get("liquidity", {}).get("usd", 0) or 0
                    if liquidity < 500: # Minimum likidite sınırı
                        continue

                    # Takip listesine ekle
                    seen_tokens.add(token_address)
                    
                    # Güvenlik analizi
                    is_safe = check_token_security(token_address)
                    
                    # Telegram Mesaj Formatı
                    dex_url = pair.get("url", f"https://dexscreener.com/{chain}/{token_address}")
                    message = (
                        f"🚀 *Yeni Meme Coin Tespit Edildi!*\n\n"
                        f"🔹 *Sembol:* `{symbol}`\n"
                        f"🌐 *Zincir:* `{chain.upper()}`\n"
                        f"💧 *Likidite:* `${liquidity:,.2f}`\n"
                        f"🛡️ *Güvenlik:* `{'İncelendi' if is_safe else 'Riskli Olabilir'}`\n\n"
                        f"🔗 [DexScreener İncele]({dex_url})\n"
                        f"📋 `{token_address}`"
                    )
                    
                    send_telegram_message(message)
            
            # Rate limit ve sistem dengesi için bekleme süresi
            time.sleep(30)
            
        except Exception as e:
            print(f"Tarama döngüsünde hata oluştu: {e}")
            time.sleep(15)

if __name__ == "__main__":
    # 1. Flask web sunucusunu arka planda (thread) başlatıyoruz
    t = threading.Thread(target=run_flask)
    t.daemon = True
    t.start()
    
    # 2. Ana akışta botun tarama döngüsünü başlatıyoruz
    bot_loop()
